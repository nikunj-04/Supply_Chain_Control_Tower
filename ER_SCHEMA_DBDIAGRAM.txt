// ─────────────────────────────────────────
// AUTH DATABASE (auth.db)
// ─────────────────────────────────────────

Table users {
  id            integer   [pk, increment]
  username      varchar   [unique, not null]
  email         varchar   [unique, not null]
  full_name     varchar   [not null]
  hashed_password varchar [not null]
  is_active     boolean   [not null, default: true]
  is_superuser  boolean   [not null, default: false]
  client_id     varchar   [note: 'Restricts data to this client for customer_user role']
  department    varchar
  phone         varchar
  created_at    datetime  [not null]
  last_login    datetime
}

Table roles {
  id            integer   [pk, increment]
  name          varchar   [unique, not null, note: 'e.g. system_admin, operations_manager']
  display_name  varchar   [not null]
  description   varchar
  is_active     boolean   [not null, default: true]
  created_at    datetime  [not null]
}

Table permissions {
  id            integer   [pk, increment]
  name          varchar   [unique, not null, note: 'e.g. wms.view, exceptions.resolve']
  display_name  varchar   [not null]
  description   varchar
  module        varchar   [not null, note: 'wms, tms, oms, billing, returns, yard, exceptions, reports']
  action        varchar   [not null, note: 'view, create, edit, delete, approve, export, assign, resolve']
  created_at    datetime  [not null]
}

Table user_roles {
  user_id       integer   [ref: > users.id, not null]
  role_id       integer   [ref: > roles.id, not null]
  indexes {
    (user_id, role_id) [pk]
  }
}

Table role_permissions {
  role_id       integer   [ref: > roles.id, not null]
  permission_id integer   [ref: > permissions.id, not null]
  indexes {
    (role_id, permission_id) [pk]
  }
}

Table user_sessions {
  id            integer   [pk, increment]
  user_id       integer   [ref: > users.id, not null]
  token         varchar   [unique, not null]
  refresh_token varchar   [unique, not null]
  ip_address    varchar
  user_agent    varchar
  created_at    datetime  [not null]
  expires_at    datetime  [not null]
  is_active     boolean   [default: true]
}

Table audit_logs {
  id            integer   [pk, increment]
  user_id       integer   [ref: > users.id]
  username      varchar
  action        varchar   [not null, note: 'login, logout, exception_resolved, charge_billed, etc.']
  resource_type varchar
  resource_id   varchar
  ip_address    varchar
  user_agent    varchar
  success       boolean   [not null]
  error_message varchar
  created_at    datetime  [not null]
}

Table exceptions {
  id                    integer  [pk, increment]
  exception_id          varchar  [unique, not null, note: 'e.g. EXC-TMS-SHIP-20001']
  exception_type        varchar  [not null, note: 'delay, inventory, quality, billing, capacity']
  severity              varchar  [not null, note: 'critical, warning, info']
  status                varchar  [not null, default: 'open', note: 'open, in_progress, resolved, dismissed']
  source_system         varchar  [not null, note: 'TMS, WMS, OMS, Billing, Returns, Yard']
  entity_type           varchar  [not null, note: 'shipment, order, inventory, invoice, return, appointment']
  entity_id             varchar  [not null, note: 'ID of the record in the source system']
  title                 varchar  [not null]
  description           text     [not null]
  impact                varchar
  customer              varchar
  location              varchar
  carrier               varchar
  days_delayed          integer
  cost_impact           float
  quantity_affected     integer
  detected_at           datetime [not null]
  expected_resolution   datetime
  resolved_at           datetime
  assigned_to           varchar  [note: 'Username of assigned user']
  assigned_at           datetime
  resolution_notes      text
  resolution_action     varchar  [note: 'reassigned, expedited, refunded, restocked, etc.']
  is_customer_notified  boolean  [default: false]
  requires_escalation   boolean  [default: false]
  is_recurring          boolean  [default: false]
}

Table exception_actions {
  id              integer  [pk, increment]
  exception_id    varchar  [ref: > exceptions.exception_id, not null]
  action_type     varchar  [not null, note: 'status_change, reassign, comment, escalate, notify']
  performed_by    varchar  [not null]
  performed_at    datetime [not null]
  notes           text
  previous_value  varchar
  new_value       varchar
}

Table exception_rules {
  id                        integer  [pk, increment]
  rule_id                   varchar  [unique, not null]
  rule_name                 varchar  [not null]
  exception_type            varchar  [not null]
  severity                  varchar  [not null]
  source_system             varchar  [not null]
  condition_field           varchar  [not null]
  condition_operator        varchar  [not null, note: 'gt, lt, eq, gte, lte']
  condition_value           varchar  [not null]
  auto_assign_to            varchar
  auto_notify               boolean  [default: true]
  auto_escalate_after_hours integer
  is_active                 boolean  [default: true]
  created_at                datetime [not null]
}
